% ---------- Option B: Neutralize Pandoc sectioning for letters ----------
% Pandoc emits \section / \subsection / \subsubsection when Markdown contains
% headings. For personal letters we want headings to behave like lightweight
% emphasis lines, not document structure.
%
% This block:
%  - ensures the commands exist even if the class/template path changes
%  - removes numbering and TOC interactions
%  - renders headings as bold (with a gentle size gradient)
%
% Place BEFORE \begin{document}.

\makeatletter

% Safe "exists?" check utility
\newcommand{\Letter@ProvideOrRenew}[2]{%
  \@ifundefined{#1}{\expandafter\newcommand\csname #1\endcsname[1]{#2}}%
                 {\expandafter\renewcommand\csname #1\endcsname[1]{#2}}%
}

% Common renderer: compact, no numbering, no TOC
\newcommand{\Letter@Heading}[2]{%
  \par\vspace{0.75\baselineskip}%
  {\noindent #1\bfseries #2\par}%
  \vspace{0.25\baselineskip}%
}

% Map sectioning to typographic headings
\Letter@ProvideOrRenew{section}{\Letter@Heading{\large}{#1}}
\Letter@ProvideOrRenew{subsection}{\Letter@Heading{\normalsize}{#1}}
\Letter@ProvideOrRenew{subsubsection}{\Letter@Heading{\small}{#1}}
\Letter@ProvideOrRenew{paragraph}{\Letter@Heading{\small}{#1}}
\Letter@ProvideOrRenew{subparagraph}{\Letter@Heading{\small}{#1}}

% Also neutralize starred variants if they exist / get called
\Letter@ProvideOrRenew{section*}{\Letter@Heading{\large}{#1}}
\Letter@ProvideOrRenew{subsection*}{\Letter@Heading{\normalsize}{#1}}
\Letter@ProvideOrRenew{subsubsection*}{\Letter@Heading{\small}{#1}}

% If anything tries to write ToC entries, drop them on the floor for letters.
\@ifundefined{addcontentsline}{}{%
  \renewcommand{\addcontentsline}[3]{}%
}

% ---------- Mythic-letter typography / artifact controls ----------
\newcommand{\ArtifactMark}{%
$if(artifact_mark)$
  {\centering\vspace*{0.06in}{\color{LetterOrnament}\normalsize $artifact_mark$\par}\vspace{0.08in}}
$endif$
}

\newcommand{\ArtifactHeader}{%
$if(artifact_header)$
  {\small\itshape
    \noindent
    \begin{minipage}[t]{0.34\textwidth}
      $if(conceived_date)$Conceived on $conceived_date$$endif$
    \end{minipage}
    \begin{minipage}[t]{0.30\textwidth}
      \centering
      $if(artifact_series)$$artifact_series$$endif$
    \end{minipage}
    \begin{minipage}[t]{0.34\textwidth}
      \raggedleft
      $if(penned_date)$Penned on $penned_date$$endif$
    \end{minipage}\par}
  \vspace{0.08in}
  {\color{LetterRule}\noindent\rule{\linewidth}{0.5pt}\par}
  \vspace{0.14in}
$endif$
}

$if(body_raggedright)$
  \RaggedRight
  \hyphenpenalty=3000

  \emergencystretch=2em
$endif$
}

% ---------- Optional stationery: letterhead ----------
% If you set letterhead_image, it will appear at the very top of page 1.
\newcommand{\MaybeLetterhead}{%
$if(letterhead_image)$
  \noindent\includegraphics[width=\textwidth,height=$letterhead_height$,keepaspectratio]{$letterhead_image$}\par\vspace{0.25in}
$endif$
}

% ---------- Optional stationery: watermark ----------
% If you set watermark_image, it will be placed behind the page content.
\newcommand{\MaybeWatermark}{%
$if(watermark_image)$
  \AddToShipoutPictureBG*{%
    \AtPageCenter{%
      \transparent{$watermark_opacity$}%
      \includegraphics[width=0.8\paperwidth,keepaspectratio]{$watermark_image$}%
    }%
  }%
$endif$
}

% ---------- Blocks: address & meta ----------
\newcommand{\SenderBlock}{%
$if(from_name)$
  {\bfseries $from_name$}\par
$endif$
$if(from_address)$
  $from_address$\par
$endif$
$if(from_email)$
  \href{mailto:$from_email$}{$from_email$}\par
$endif$
$if(from_phone)$
  $from_phone$\par
$endif$
}

\newcommand{\RecipientBlock}{%
$if(to_name)$
  {\bfseries $to_name$}\par
$endif$
$if(to_address)$
  $to_address$\par
$endif$
}

\newcommand{\LetterMeta}{%
$if(date)$
  {\small $date$}\par
$endif$
$if(subject)$
  \vspace{0.15in}
  {\bfseries Subject: $subject$}\par
$endif$
}

% ---------- Signature ----------
\newcommand{\SignatureBlock}{%
$if(signature_style)$
  \vspace{0.35in}
$else$
  \vspace{0.25in}
$endif$
$if(closing)$
  $if(signature_style)$
    {\itshape $closing$}\par
  $else$
    $closing$\par
  $endif$
$endif$
\vspace{0.16in}
$if(signature_image)$
  \includegraphics[scale=$signature_scale$]{$signature_image$}\par
  \vspace{0.05in}
$endif$
$if(signature_name)$
  {\bfseries $signature_name$}\par
$endif$
}
